範圍

Spec 的 Phase 1 有五個步驟，這份計畫只涵蓋第 1、2 步 —— 也就是地基。

不在這份計畫裡：行為按能力搬家（憑證 → 啟動／結束 → shell → phantom）、 24 處 Tool.allCases 的遷移、刪除 enum、把 Codex 與 Gemini 移到各自 repo。

那些屬於另一份計畫，等這個地基存在、介面被真實壓過一輪之後再寫。 現在寫等於在猜一個還沒有人用過的介面長什麼樣。

這份計畫本身能產出可運作的軟體：orrery 行為與今天完全相同，registry 就位在 enum 旁邊， 旗標的危險已經關閉。

全域約束

  • AIToolKit 不得引用任何 orrery 型別。它是第三方可以單獨依賴、不必拖進 orrery 的套件。
  • 整份計畫期間 AIToolKit 停在 0.x。允許 breaking change，1.0 起才遵守 semver。
  • on-disk 格式不變。metadata.json 保持 "tool": "claude"；帳號路徑保持 accounts/<tool>/<id>/。
  • CLI 介面不變。--claude / --codex / --gemini 繼續可用。
  • 既有安裝不得需要重新登入或遷移。
  • 測試用 swift-testing 而非 XCTest。碰到 ORRERY_HOME 或真實 home 的測試必須包在 withIsolatedHome { … } 裡。
  • 這份計畫不搬動任何行為離開 Tool enum。

Task 1

MigrationFlag —— per-tool 涵蓋記錄

為什麼這件事排第一

四個一次性遷移各自只寫一個全域的「做完了」標記。一旦 Tool.allCases 被 registry 取代， 在那一刻沒有註冊的工具會被跳過 ——而旗標照樣寫下去，於是那台機器上它永遠不會被遷移。 這個 task 建造替代品，Task 2 才套用。

相容性在這裡很要緊：既有安裝的旗標檔內容是 v1\n 或 v3\n。 那些必須被讀成「已經涵蓋一切」，否則每個升級的使用者都會重跑已經跑過的遷移。

Step 3 · 實作

檔案格式是版本行加上每行一個 tool id。沒有 id 行的檔案就是舊標記，讀成涵蓋一切。

public struct MigrationFlag {
    public enum Coverage: Equatable {
        case absent            // 從未執行
        case legacyCoversAll   // per-tool 之前的舊標記，視為涵蓋所有工具
        case ids(Set<String>)
    }

    private static let formatVersion = "v2"
    public let url: URL

    public func coverage() -> Coverage {
        guard let data = try? Data(contentsOf: url),
              let text = String(data: data, encoding: .utf8)
        else { return .absent }

        let ids = text.split(separator: "\n")
            .map { $0.trimmingCharacters(in: .whitespaces) }
            .filter { !$0.isEmpty && $0 != Self.formatVersion && !isVersionToken($0) }

        return ids.isEmpty ? .legacyCoversAll : .ids(Set(ids))
    }

    public func pending(among candidates: Set<String>) -> Set<String> {
        switch coverage() {
        case .absent:          return candidates
        case .legacyCoversAll: return []
        case .ids(let done):   return candidates.subtracting(done)
        }
    }

    /// union 而非覆寫：為了一個晚註冊的工具而重跑，不該抹掉已處理過的記錄。
    public func markCovered(_ ids: Set<String>) throws {
        var all = ids
        if case .ids(let existing) = coverage() { all.formUnion(existing) }
        let body = ([Self.formatVersion] + all.sorted()).joined(separator: "\n") + "\n"
        try Data(body.utf8).write(to: url, options: .atomic)
    }

    /// v1、v3 …… 是舊標記行，不是 tool id。
    private func isVersionToken(_ s: String) -> Bool {
        s.first == "v" && s.dropFirst().allSatisfy(\.isNumber) && s.count > 1
    }
}

Task 3

AIToolKit 套件與 AITool 型別

兩個形狀決定，都是 spec 判準的實際應用

configDirectoryName 是名字不是 URL。解析它需要 home 目錄， 而用哪個 home 是 orrery 的決定 —— userHomeURL() 可被 ORRERY_USER_HOME 覆寫正是為了讓測試能重導。放 URL 會把那個政策拖進 framework。

configDirEnvVar 是 Optional，這修掉了一個現存的謊。 現在 Tool.gemini.envVarName 回傳 "GEMINI_CONFIG_DIR"， 但 gemini-cli 完全無視它、只讀 $HOME/.gemini —— 設定它正是 orrery add --gemini 寫進使用者真實設定的那個 bug。 nil 誠實地陳述事實；要不要改用 HOME wrapper 是宿主的決定。

public struct AITool: Sendable, Hashable, Codable {
    public let id: String
    public let displayName: String
    public let configDirectoryName: String   // ".claude" —— 名字，不是 URL
    public let configDirEnvVar: String?      // nil = 這個工具沒有這種變數
    public let authLoginCommand: [String]?   // nil = 首次互動啟動時認證
    public let installCommand: [String]?
    public let sessionSubdirectories: [String]
    public let ansiColor: String

    public var supportsSetup: Bool { installCommand != nil }
    public var coloredTag: String { "\(ansiColor)[\(id)]\u{1B}[0m" }

    // 身分即 id
    public static func == (lhs: AITool, rhs: AITool) -> Bool { lhs.id == rhs.id }
    public func hash(into hasher: inout Hasher) { hasher.combine(id) }

    // 編碼成裸 id，讓宿主既有的 on-disk 格式原封不動
    public func encode(to encoder: any Encoder) throws {
        var container = encoder.singleValueContainer()
        try container.encode(id)
    }
}

///

public final class AIToolRegistry: @unchecked Sendable {
    public static let shared = AIToolRegistry()

    private let lock = NSLock()
    private var storage: [String: AITool] = [:]

    public var all: [AITool] { /* lock; storage.values.sorted { $0.id < $1.id } */ }

    public var isEmpty: Bool { /* … */ }
    public func reset() { /* 測試用 */ }
}

extension Tool {
    /// 用 AIToolKit 的語彙描述這個工具。
    ///
    /// 是橋接不是取代：enum 沒有被移除任何東西，呼叫點可以一次搬一個。
    /// 等到沒有人再讀它，enum 才刪掉。
    public var aiTool: AITool {
        AITool(id: rawValue, displayName: displayName,
               configDirectoryName: defaultConfigDir.lastPathComponent,
               configDirEnvVar: bridgedConfigDirEnvVar,
               authLoginCommand: authLoginCommand, installCommand: installCommand,
               sessionSubdirectories: sessionSubdirectories, ansiColor: ansiColor)
    }

    /// envVarName 對每個 case 都回傳字串，包括 gemini —— 但 gemini-cli 無視
    /// GEMINI_CONFIG_DIR、只讀 $HOME/.gemini。設定它正是 orrery add --gemini
    /// 寫進使用者真實設定的那個 bug。橋接不把這個主張帶過去；
    /// enum 保留舊值直到它的呼叫者清空為止。
    private var bridgedConfigDirEnvVar: String? {
        switch self {
        case .claude, .codex: return envVarName
        case .gemini:         return nil
        }
    }
}
